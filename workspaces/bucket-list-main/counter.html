카운터 컴포넌트

기본 카운터

범위 제한 카운터 (0 ~ 10)

스텝 카운터 (+5씩)

사용 방법

// 기본 카운터 생성
const counter = new Counter({
  initialValue: 0,
  min: -Infinity,
  max: Infinity,
  step: 1,
  onChange: (value) => console.log('현재값:', value)
});

// 값 증가/감소
counter.increment();
counter.decrement();

// 값 설정/리셋
counter.setValue(5);
counter.reset();

// DOM에 UI 렌더링
const counterUI = new CounterUI('containerId', {
  initialValue: 0,
  min: 0,
  max: 10,
  step: 1
});
